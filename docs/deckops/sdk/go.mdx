```bash
go get github.com/deckops/deckops/sdks/go
```

```go
package main

import (
  "context"
  "os"

  deckops "github.com/deckops/deckops/sdks/go"
)

func main() {
  ctx := context.Background()
  deck, err := deckops.New(ctx, deckops.ClientOptions{
    APIKey: os.Getenv("DECKOPS_API_KEY"),
  })
  if err != nil { panic(err) }

  task, err := deck.ConvertPptToPDF(ctx, deckops.TaskShortcutParams{
    Files: []deckops.TaskUploadInput{{
      Input: deckops.UploadInput{Path: "./slides.pptx"},
    }},
  })
  if err != nil { panic(err) }

  _, err = deck.Tasks.Wait(ctx, task.ID, deckops.WaitForTaskOptions{})
  if err != nil { panic(err) }
}
```

The Go SDK provides typed task shortcuts, upload support for paths, byte slices and readers, guest UUID persistence, SSE or polling waits, parse routing, and structured-result-to-Markdown converters.
